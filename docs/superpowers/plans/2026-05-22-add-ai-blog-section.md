> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a new top-level `/blogs` section to Agentic AI Wiki — chronological feed, tag pages, bilingual en/zh — and ship the first post: a deep technical comparison of OpenClaw, OpenHuman, and Hermes Agent.

**Architecture:** Mirror the existing one-file-per-entry content pattern (already used by the changelog) for posts under `src/content/blogs/posts/<YYYY-MM-DD>-<slug>.ts`; body fragments are body-only HTML at `src/content/blogs/{en,zh}/<slug>.html`; diagrams are co-located SVGs under `public/blogs/<slug>/`. **Tech Stack:** Astro 4 (static), TypeScript, Node test runner (`node --test --experimental-strip-types`), Pagefind, Vercel build pipeline. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-05-22-ai-blog-section-design.md`

**Branch / worktree:** `feature/add-ai-blog-section-2026-05-22` in `.worktrees/add-ai-blog-section`. Run all commands from that worktree directory.

- [ ] **Step 2: Create `src/content/blogs/manifest.ts`**

```ts
// Aggregator for the AI Blog posts.
//
// Each POST lives in its own file under `./posts/<YYYY-MM-DD>-<slug>.ts` —
// one file per post — so concurrent PRs adding posts never collide. Vite's
// `import.meta.glob` resolves all post files at build time; posts are sorted
// by filename (which is date-prefixed) DESCENDING so the newest post is first.
import type { BlogPost } from './types';

export type { BlogPost, Locale, Localized } from './types';
export { L } from './types';

interface PostModule { default: BlogPost }
const modules = import.meta.glob<PostModule>('./posts/*.ts', { eager: true });

/** Posts sorted newest-first by filename (date prefix), slug as tiebreaker within a day. */
export const POSTS: BlogPost[] = Object.entries(modules)
  .map(([path, m]) => ({ post: m.default, file: path.split('/').pop()! }))
  .sort((a, b) => b.file.localeCompare(a.file))
  .map(({ post }) => post);

/** Lookup a post by its slug. */
export function postBySlug(slug: string): BlogPost | undefined {
  return POSTS.find(p => p.slug === slug);
}

/** Posts that include the given tag (case-sensitive on lowercase kebab-case tags). */
export function postsByTag(tag: string): BlogPost[] {
  return POSTS.filter(p => p.tags.includes(tag));
}

/** Sorted, deduplicated list of all tags across all posts. */
export function allTags(): string[] {
  const s = new Set<string>();
  for (const p of POSTS) for (const t of p.tags) s.add(t);
  return [...s].sort();
}
```

```js
// scripts/__tests__/blogs.test.mjs
// Validates every blog post file under src/content/blogs/posts/ for the
// one-file-per-post convention and that every registered post has bilingual
// HTML fragments. Mirrors changelog.test.mjs.
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readdirSync, existsSync } from 'node:fs';
import { fileURLToPath, pathToFileURL } from 'node:url';

const postsDir = fileURLToPath(new URL('../../src/content/blogs/posts/', import.meta.url));
const enDir    = fileURLToPath(new URL('../../src/content/blogs/en/',    import.meta.url));
const zhDir    = fileURLToPath(new URL('../../src/content/blogs/zh/',    import.meta.url));
const files    = readdirSync(postsDir).filter(f => f.endsWith('.ts')).sort();

const SLUG_RE = /^[a-z0-9](?:[a-z0-9-]*[a-z0-9])?$/;
const FILE_RE = /^(\d{4}-\d{2}-\d{2})-([a-z0-9](?:[a-z0-9-]*[a-z0-9])?)\.ts$/;

test('blogs: at least one post file exists', () => {
  assert.ok(files.length > 0, `no post files under ${postsDir}`);
});

test('blogs: filenames are <YYYY-MM-DD>-<slug>.ts', () => {
  for (const f of files) assert.match(f, FILE_RE, `bad filename: ${f}`);
});

test('blogs: every post is well-shaped, bilingual, and has en+zh fragments', async () => {
  const seenSlugs = new Set();
  for (const f of files) {
    const m = await import(pathToFileURL(postsDir + f).href);
    const p = m.default;
    assert.ok(p, `${f}: missing default export`);

    const fileMatch = f.match(FILE_RE);
    assert.ok(fileMatch, `${f}: filename did not match expected pattern`);
    const [, fileDate, fileSlug] = fileMatch;

    assert.match(p.date, /^\d{4}-\d{2}-\d{2}$/, `${f}: invalid date ${p.date}`);
    assert.equal(p.date, fileDate, `${f}: filename date prefix (${fileDate}) must equal post.date (${p.date})`);

    assert.match(p.slug, SLUG_RE, `${f}: invalid slug ${p.slug}`);
    assert.equal(p.slug, fileSlug, `${f}: filename slug (${fileSlug}) must equal post.slug (${p.slug})`);
    assert.ok(!seenSlugs.has(p.slug), `${f}: duplicate slug ${p.slug}`);
    seenSlugs.add(p.slug);

    assert.ok(p.title?.en?.trim() && p.title?.zh?.trim(),     `${f}: title must be bilingual non-empty`);
    assert.ok(p.summary?.en?.trim() && p.summary?.zh?.trim(), `${f}: summary must be bilingual non-empty`);

    assert.ok(Array.isArray(p.tags) && p.tags.length > 0, `${f}: tags must be a non-empty array`);
    for (const t of p.tags) {
      assert.equal(typeof t, 'string', `${f}: tag must be a string`);
      assert.match(t, SLUG_RE, `${f}: tag "${t}" must be lowercase kebab-case`);
    }

    if (p.author !== undefined) {
      assert.ok(p.author.en?.trim() && p.author.zh?.trim(), `${f}: author override must be bilingual non-empty`);
    }

    assert.ok(existsSync(`${enDir}${p.slug}.html`), `${f}: missing en fragment for ${p.slug}`);
    assert.ok(existsSync(`${zhDir}${p.slug}.html`), `${f}: missing zh fragment for ${p.slug}`);
  }
});
```

// Renders the post header (title, date, author, reading-time, tags) + body + footer
// (latest posts / more tagged X). Emits BlogPosting JSON-LD always. If the post

// Reading time: explicit override else estimate from word count (Chinese counted by
// non-whitespace runs; English by whitespace splits).
function estimateReadingTime(html: string, loc: Locale): number {
  const text = html.replace(/<[^>]+>/g, ' ').replace(/\s+/g, ' ').trim();
  if (loc === 'zh') {
    // Strip whitespace and count remaining characters; ~400 char/min as the common heuristic.
    const chars = text.replace(/\s+/g, '').length;
    return Math.max(1, Math.round(chars / 400));
  }
  const words = text ? text.split(/\s+/).length : 0;
  return Math.max(1, Math.round(words / 225));
}

};

// Tolerant: silent no-op if the section is absent or malformed.
function extractFaq(html: string): { question: string; answer: string }[] {
  const section = html.match(/<section[^>]*class="[^"]*\bfaq\b[^"]*"[^>]*>([\s\S]*?)<\/section>/i);
  if (!section) return [];
  const inner = section[1];
  const pairs: { question: string; answer: string }[] = [];
  const re = /<h3[^>]*>([\s\S]*?)<\/h3>\s*<p[^>]*>([\s\S]*?)<\/p>/gi;
  let m: RegExpExecArray | null;
  while ((m = re.exec(inner)) !== null) {
    const question = m[1].replace(/<[^>]+>/g, '').replace(/\s+/g, ' ').trim();
    const answer   = m[2].replace(/<[^>]+>/g, '').replace(/\s+/g, ' ').trim();
    if (question && answer) pairs.push({ question, answer });
  }
  return pairs;
}

  mainEntity: faq.map(({ question, answer }) => ({
    '@type': 'Question',
    name: question,
    acceptedAnswer: { '@type': 'Answer', text: answer },
  })),
} : null;

const safeJsonLd = (obj: unknown) =>
  JSON.stringify(obj).replace(/</g, String.fromCharCode(92) + 'u003c');

  <article class="wrap" style="max-width:760px;padding:28px 20px;">
    <div class="breadcrumb" data-pagefind-ignore>
      <a href={localizeHref('/blogs', locale)}>{t.blog.h1}</a> / {title}
    </div>
    <header style="margin:18px 0 26px;">
      <h1 style="font-family:'Fraunces',serif;font-size:34px;line-height:1.2;margin:0 0 12px;">{title}</h1>
      <div style="color:var(--muted);font-size:14px;display:flex;flex-wrap:wrap;gap:12px;align-items:center;">
        <time datetime={post.date}>{t.blog.postedOn} {post.date}</time>
        <span aria-hidden="true">·</span>
        <span>{t.blog.byline} {author}</span>
        <span aria-hidden="true">·</span>
        <span>{readingTimeMin} {t.blog.readingTime}</span>
      </div>
      <ul style="list-style:none;padding:0;margin:10px 0 0;display:flex;flex-wrap:wrap;gap:6px;">
        {post.tags.map(tag => (
          <li><a href={localizeHref(`/blogs/tag/${tag}`, locale)}
                 aria-label={`${t.blog.taggedWith}: ${tag}`}
                 style="display:inline-block;padding:2px 8px;border:1px solid var(--border-soft);border-radius:999px;font-size:12px;color:var(--muted);text-decoration:none;">{tag}</a></li>
        ))}
      </ul>
    </header>

    <div class="blog-body" set:html={body} />

    <footer data-pagefind-ignore style="margin-top:48px;border-top:1px solid var(--border-soft);padding-top:24px;display:grid;grid-template-columns:repeat(auto-fit,minmax(280px,1fr));gap:24px;">

          <h2 style="font-family:'Fraunces',serif;font-size:18px;margin:0 0 10px;">{t.blog.taggedWith}: {primaryTag}</h2>
          <ul style="list-style:none;padding:0;margin:0;display:grid;gap:8px;">

            ))}
          </ul>

        <h2 style="font-family:'Fraunces',serif;font-size:18px;margin:0 0 10px;">{t.blog.latestPosts}</h2>
